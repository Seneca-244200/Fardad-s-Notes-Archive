Binary operators with no side effect
 A @ B // binary operator (two operands) 
 a = 2;
 b = 3;
 cout << a + b << endl; // + is binary op. 
 // output 5
 cout << a << " " << b << endl;
 // output 2 3    + does not have a side effect
 cout << a < b << endl
 // output 1 
 cout << a << " " << b << endl;
 // output 2 3    < does not have a side effect
 
Binary operators with side-effect
 a = 2;
 b = 3
 cout << a += b << endl;
 // output 5
 cout << a << " " << b << endl;
 // output 5 3    += does have a side effect

Unary operators:

   a = 2
